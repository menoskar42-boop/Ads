WITH base AS (SELECT ce.id, ce.stock_tech_name AS tech, CASE WHEN btrim(ce.work_order_type)='صيانة' THEN 'maint' ELSE 'install' END AS type, NULLIF(ce.cable_quantity,'')::numeric AS qty,ce.source,ce.stock_date,ce.wo_ref,ce.phone_local FROM cable_entries ce WHERE ce.stock_tech_name IS NOT NULL AND ce.stock_date >= CASE WHEN btrim(ce.work_order_type)='صيانة' THEN DATE '2026-09-29' ELSE DATE '2026-10-06' END) SELECT 'qty_gt_300' AS check_name,count(*)::int AS count,COALESCE(SUM(qty),0)::float AS qty FROM base WHERE qty>300 UNION ALL SELECT 'sheet_qty_50',count(*)::int,COALESCE(SUM(qty),0)::float FROM base WHERE source='sheet' AND qty=50 UNION ALL SELECT 'closed_before_cutoff',count(*)::int,COALESCE(SUM(b.qty),0)::float FROM base b JOIN work_orders w ON w.id=b.wo_ref WHERE (w.close_date AT TIME ZONE 'Africa/Cairo')::date < CASE WHEN b.type='maint' THEN DATE '2026-09-29' ELSE DATE '2026-10-06' END UNION ALL SELECT 'duplicate_wo_ref_entries',count(*)::int,COALESCE(SUM(qty),0)::float FROM base WHERE wo_ref IN(SELECT wo_ref FROM base WHERE wo_ref IS NOT NULL GROUP BY wo_ref HAVING count(*)>1) UNION ALL SELECT 'install_not_success',count(*)::int,COALESCE(SUM(b.qty),0)::float FROM base b JOIN work_orders w ON w.id=b.wo_ref WHERE b.type='install' AND w.close_category IS DISTINCT FROM 'Success'

### إثبات فرق شكل الهاتف من غير كشف الرقم

WITH b AS(SELECT wo_ref,phone_local,work_order_type FROM cable_entries WHERE stock_tech_name IS NOT NULL AND stock_date>=CASE WHEN btrim(work_order_type)='صيانة' THEN DATE '2026-09-29' ELSE DATE '2026-10-06' END), d AS(SELECT wo_ref,count(*) AS n,count(DISTINCT phone_local) AS phone_variants,count(DISTINCT regexp_replace(phone_local,'[^0-9]','','g')) AS digit_variants,count(DISTINCT work_order_type) AS type_variants FROM b WHERE wo_ref IS NOT NULL GROUP BY wo_ref HAVING count(*)>1) SELECT phone_variants,digit_variants,type_variants,COUNT(*)::int AS orders FROM d GROUP BY 1,2,3
